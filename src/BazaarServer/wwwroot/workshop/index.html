常用 CLI 指令（tools/mod_sdk_cli.py）

            
# 1. 初始化标准 Mod 骨架
python tools/mod_sdk_cli.py new my-forge --name "星港锻炉" --author "你的名字"

# 2. 检索 3,368 原版白名单卡并一键克隆 / 生成原位补丁
python tools/mod_sdk_cli.py clone-card --search "Kukri" --type Item --size Small
python tools/mod_sdk_cli.py clone-card "Kukri" --mod my-forge --as storm_kukri --mode clone
python tools/mod_sdk_cli.py clone-card "Kukri" --mod my-forge --mode patch

# 3. 跑满 10 道防 Bug 铁闸 + 50 场真实战斗体检
python tools/mod_sdk_cli.py validate mods/my-forge --strict --offline-sim
python tools/mod_sdk_cli.py sim-test mods/my-forge --matches 50

# 4. 打包分享 & 安全检疫解包
python tools/mod_sdk_cli.py pack mods/my-forge -o dist/mods/my-forge.bazaarmod.zip
python tools/mod_sdk_cli.py unpack dist/mods/my-forge.bazaarmod.zip --into mods/
